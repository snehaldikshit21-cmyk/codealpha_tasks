import java.util.Scanner;

public class AIChatbot {

    public static String getResponse(String input) {

        input = input.toLowerCase().trim();

        if (input.contains("hello") || input.contains("hi") || input.contains("hey")) {
            return "Hello! How can I help you?";
        }

        if (input.contains("your name") || input.contains("who are you")) {
            return "I am JavaBot, your AI chatbot.";
        }

        if (input.contains("how are you")) {
            return "I am doing great! Thank you for asking.";
        }

        if (input.contains("java")) {
            return "Java is a popular object-oriented programming language.";
        }

        if (input.contains("artificial intelligence") || input.contains("ai")) {
            return "AI allows computers to perform tasks that normally require human intelligence.";
        }

        if (input.contains("nlp") || input.contains("natural language")) {
            return "NLP stands for Natural Language Processing.";
        }

        if (input.contains("help")) {
            return "I can answer questions about Java, AI and NLP.";
        }

        if (input.contains("thank")) {
            return "You're welcome!";
        }

        if (input.contains("bye") || input.contains("exit")) {
            return "Goodbye! Have a great day.";
        }

        return "Sorry, I don't understand. Please try another question.";
    }

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.println("================================");
        System.out.println("        JAVA AI CHATBOT");
        System.out.println("================================");
        System.out.println("Bot: Hello! I am JavaBot.");
        System.out.println("Bot: Type 'bye' to exit.");

        while (true) {

            System.out.print("\nYou: ");
            String input = scanner.nextLine();

            String response = getResponse(input);

            System.out.println("Bot: " + response);

            if (input.toLowerCase().contains("bye") ||
                input.toLowerCase().contains("exit")) {
                break;
            }
        }

        scanner.close();
    }
}
